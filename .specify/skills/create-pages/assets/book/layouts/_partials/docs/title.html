{{- /* Nav / link label, with a first-

fallback. Overrides the theme partial. Documents in this space carry no YAML front matter, so .Title and .LinkTitle are empty and the theme would fall back to a humanized file name ("Documentation Model" for a page whose real H1 is "Documentation Model (What & Why)", "Notes" for a Chinese-titled page). The H1 is read from .RawContent, not .Content, because the theme's heading render hook appends an anchor link — plainifying the rendered

yields a trailing "#". Machine-generated section pages get their label from the managed `title` cascade in hugo.toml instead. */ -}} {{- $title := "" -}} {{- if .LinkTitle -}} {{- $title = .LinkTitle -}} {{- else if .Title -}} {{- $title = .Title -}} {{- else -}} {{- with findRE `(?m)^#[ \t]+(.+?)[ \t]*$` .RawContent 1 -}} {{- $title = index . 0 | strings.TrimPrefix "#" | strings.TrimSpace | replaceRE "[`*_]" "" -}} {{- end -}} {{- end -}} {{- if not $title -}} {{- with .File -}} {{- $base := cond $.IsSection (path.Base .Dir) .BaseFileName -}} {{- $title = replaceRE `[-_]+` " " $base | strings.FirstUpper -}} {{- end -}} {{- end -}} {{- return $title -}}
